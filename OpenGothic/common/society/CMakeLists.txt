file(GLOB SOCIETY_CORE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_library(society-core STATIC ${SOCIETY_CORE_SOURCES})
set_property(TARGET society-core PROPERTY INCLUDE_DIRECTORIES "")
set_property(TARGET society-core PROPERTY POSITION_INDEPENDENT_CODE ON)
target_include_directories(society-core SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/rapidjson/include")
target_include_directories(society-core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(society-core PUBLIC cxx_std_20)
if(NOT MSVC)
  target_compile_options(society-core PRIVATE -Wall -Wconversion -Werror -ffp-contract=off)
endif()
