cmake_minimum_required(VERSION 3.20)
project(cloth-census LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ZK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_ASAN OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SQUISH_WITH_OPENMP OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  set(BUILD_SQUISH_WITH_SSE2 OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(../../OpenGothic/lib/ZenKit zenkit)
add_executable(cloth-census census.cpp)
target_link_libraries(cloth-census PRIVATE zenkit)
